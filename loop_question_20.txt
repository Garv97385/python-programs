# Write a Python program that asks the user to enter a number n.
# The program should then print the multiplication table of that number from 1 to 10 using a for loop.

n = int(input("Enter a number : "))
print(f"Here is your table of {n} :  ")

for i in range(1,11):
    print(f"{n}  x {i} = {n*i}")
