# Find the sum of odd elements only.

n = int(input("Enter the number of elements : "))
arr = []
odd_arr = []

for i in range(n):
    num = int(input("Enter the number : "))
    arr.append(num)
    if num % 2 != 0:
        odd_arr.append(num)

odd = sum(odd_arr)

print("The array is",arr)
print("The sum of odd numbers in array :",odd)
