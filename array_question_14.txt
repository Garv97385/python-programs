# Create a new array containing only even elements.

n = int(input("Enter the number of elements : "))
arr = []
even_arr = []

for i in range(n):
    num = int(input("Enter the number : "))
    arr.append(num)

for j in arr:
    if j % 2 == 0:
        even_arr.append(j)

print("Original array :", arr)
print("Even's array :", even_arr)
