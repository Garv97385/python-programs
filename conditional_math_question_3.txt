# Take marks (0–100) and print the corresponding grade (A/B/C/D/F)

marks = float(input("Enter your marks (0-100): "))

if marks > 90 :
    print ("Grade: A")
elif marks > 80 :
    print ("Grade : B")
elif marks > 60 :
    print ("Grade : C")
elif marks > 40 :
    print ("Grade : D")
else :
    print ("Grade : F")
