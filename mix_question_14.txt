# Take age inputs and count how many are adults, minors, seniors.

def count_age_groups(ages):
    minors = adults = seniors = 0
    
    for age in ages:
        if age < 18:
            minors += 1
        elif 18 <= age < 60:
            adults += 1
        else:  # age >= 60
            seniors += 1
    
    return minors, adults, seniors

# Example usage
ages = [12, 25, 17, 45, 61, 80, 30, 15, 70]
minors, adults, seniors = count_age_groups(ages)

print(f"Minors: {minors}")
print(f"Adults: {adults}")
print(f"Seniors: {seniors}")
