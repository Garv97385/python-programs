# Shift all zeros to the end of the array. 

def shift_zeros(arr):
    non_zero_index = 0
    
    
    for num in arr:
        if num != 0:
            arr[non_zero_index] = num
            non_zero_index += 1
    
    while non_zero_index < len(arr):
        arr[non_zero_index] = 0
        non_zero_index += 1
    
    return arr

arr = [0, 1, 0, 3, 12]
print(shift_zeros(arr))
