# Take n elements and print only those greater than a given value k. 

n = int(input("Enter number of elements: "))
k = int(input("Enter the value of k: "))


arr = []
for i in range(n):
    num = int(input(f"Enter element {i+1}: "))
    arr.append(num)
print(f"Elements greater than {k}:")
for num in arr:
    if num > k:
        print(num)
