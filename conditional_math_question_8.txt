# Take an alphabet character and check if it lies between ‘a’ and ‘m’ or ‘n’ and ‘z’. 

print("please enter a small alphabet character otherwise the program will not work ")

char = input("Enter a alphabet character : ")
if char >= "a"  and  char <= "m " :
    print("The character lies b/w a and m ")
elif char >= "n" and char <= "z" :
    print("The character lies b/w n and z")
