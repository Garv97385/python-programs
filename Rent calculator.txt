# Rent calculator

rent = int(input("Enter the total rent of hostle/flat : "))
food = int(input("Enter the total food/snack ordered : "))
electricity = int(input("Enter the total spend in electricity : "))
charge_per_unit = int(input("Enter the charge per unit : "))
persons = int(input("Enter the person living in hostle/flat : "))

total_electricity_bill = electricity * charge_per_unit
output = (rent + food + total_electricity_bill) // persons

print(f"Each person will pay : {output}")
