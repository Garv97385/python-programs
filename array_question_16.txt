# Replace all even numbers with 1 and all odd with 0

arr = []
n = int(input("Enter the number of element : "))

for i in range(n):
    arr.append (int(input("Enter the number : ")))

for i in range(len(arr)):
    if arr[i] % 2 == 0 :
        arr[i] = 1
    elif arr[i] % 2 != 0 :
        arr[i] = 0
print(arr)
